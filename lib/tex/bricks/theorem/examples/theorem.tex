\section{Calculus}
\begin{theorem}[Mean value theorem]\label{thm:mvt}
If $f$ is continuous on $[a,b]$ and differentiable on $(a,b)$, then some
$c\in(a,b)$ satisfies
\[
  f'(c)=\frac{f(b)-f(a)}{b-a}.
\]
\end{theorem}
