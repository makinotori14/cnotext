\section{Divisibility}
\begin{theorem}\label{thm:division}
For $a\in\mathbb Z$ and $b>0$, there are unique $q,r\in\mathbb Z$ such that
$a=bq+r$ and $0\le r<b$.
\end{theorem}
\begin{corollary}
Every integer is either even or odd.
\end{corollary}
