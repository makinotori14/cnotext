\begin{solution}[Solution by substitution]
Set $u=x^2$. Then
\begin{align*}
u^2-5u+4&=0,\\
(u-1)(u-4)&=0.
\end{align*}
Thus $x^2\in\{1,4\}$.
\answer{$x\in\{-2,-1,1,2\}$.}
\end{solution}
